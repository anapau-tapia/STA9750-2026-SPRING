read_olympedia <- function(url, ...){
  if(stringr::str_starts(url, "/")) {
    url <- stringr::str_remove(url, "^/")
  }

  cache_dir <- fs::path("data", "mp04", "olympedia_cache")
  if(!fs::dir_exists(cache_dir)) fs::dir_create(cache_dir, recurse = TRUE)

  cache_path <- fs::path(
    cache_dir,
    url |> as.character() |> stringr::str_replace_all("/", "_")
  )

  if(!fs::file_exists(cache_path)){
    src_url <- paste0("https://www.olympedia.org/", url)
    slow_download(src_url, cache_path, mode = "wb")
  }

  rvest::read_html(cache_path)
}

# Builds a table of Olympic editions with year, city, season, and edition ID.
read_editions <- function() {
  pg <- read_olympedia("editions")

  tables <- pg |> html_elements("table")

  parse_editions_table <- function(tbl, season_label) {
    rows <- tbl |> html_elements("tr")

    map_dfr(rows, function(row) {
      tds <- row |> html_elements("td")
      if(length(tds) < 8) return(NULL)

      edition_link <- row |> html_element("a[href^='/editions/']")
      edition_url <- if(length(edition_link) == 0) NA_character_ else html_attr(edition_link, "href")
      edition_id  <- str_extract(edition_url, "\\d+$") |> as.integer()

      olympiad_roman <- tds[[1]] |> html_text2() |> str_trim()
      year_txt <- tds[[2]] |> html_text2() |> str_trim()
      city <- tds[[3]] |> html_text2() |> str_trim()
      note <- tds[[8]] |> html_text2() |> str_squish()

      year <- suppressWarnings(as.integer(year_txt))

      flag_img <- tds[[4]] |> html_element("img")
      flag_src <- if(length(flag_img) == 0) NA_character_ else html_attr(flag_img, "src")
      host_country_code <- str_extract(flag_src, "[A-Z]{3}(?=\\.png$)")

      if(is.na(edition_id) || is.na(year) || olympiad_roman == "") return(NULL)

      tibble(
        season = season_label,
        year = year,
        city = city,
        olympiad_roman = olympiad_roman,
        olympiad_id = roman_to_integer(olympiad_roman),
        edition_id = edition_id,
        edition_url = edition_url,
        host_country_code = host_country_code,
        note = note
      )
    })
  }

  bind_rows(
    parse_editions_table(tables[[1]], "Summer"),
    parse_editions_table(tables[[2]], "Winter")
  ) |>
    distinct(edition_id, .keep_all = TRUE) |>
    filter(!str_detect(str_to_lower(note), "not held")) |>
    select(-note) |>
    arrange(year, season)
}

#Join the edition-level data to the list of disciplines available for each edition.


read_edition_disciplines <- function(edition_url) {
  pg <- read_olympedia(edition_url)

  sport_table <- pg |>
    html_element(xpath = "//h2[normalize-space()='Medal Disciplines']/following-sibling::table[1]")

  if(length(sport_table) == 0) {
    return(tibble(
      discipline = character(),
      discipline_url = character(),
      discipline_code = character()
    ))
  }

  links <- sport_table |> html_elements("a")

  tibble(
    discipline = html_text2(links) |> str_squish(),
    discipline_url = html_attr(links, "href")
  ) |>
    filter(str_detect(discipline_url, "/sports/")) |>
    mutate(
      discipline_code = str_extract(discipline_url, "(?<=/sports/)[A-Z0-9]+")
    ) |>
    distinct()
}

# This chunk combines all of the discipline medal tables into one master Olympedia medal dataset.
read_discipline_medal_table <- function(discipline_url) {
  pg <- read_olympedia(discipline_url)

  medal_table_node <- pg |>
    html_element(xpath = "//h2[normalize-space()='Medal table']/following-sibling::table[1]")

  if(length(medal_table_node) == 0) {
    return(tibble(
      competitor = character(),
      country_code = character(),
      gold = integer(),
      silver = integer(),
      bronze = integer(),
      total = integer()
    ))
  }

  medal_tbl <- medal_table_node |> html_table(fill = TRUE)

  # At least 6 columns: competitor, 3‑letter code, then Gold, Silver, Bronze, Total
  if(ncol(medal_tbl) < 6) {
    return(tibble(
      competitor = character(),
      country_code = character(),
      gold = integer(),
      silver = integer(),
      bronze = integer(),
      total = integer()
    ))
  }

  tibble(
    competitor = as.character(medal_tbl[[1]]),
    country_code = as.character(medal_tbl[[2]]),
    gold = suppressWarnings(as.integer(medal_tbl[[3]])),
    silver = suppressWarnings(as.integer(medal_tbl[[4]])),
    bronze = suppressWarnings(as.integer(medal_tbl[[5]])),
    total = suppressWarnings(as.integer(medal_tbl[[6]]))
  ) |>
    filter(!is.na(country_code), country_code != "")
}

Host Nation Effect

To analyze the Host Nation Effect, I took 6 different countries from different continents: France, Japan, Brazil, Great Britain, China and Australia, to analyze their historical performance from 1990 - 2024, and highlight their hosting year. This will give us a clear picture of their ‘normal’ performance when these countries are abroad, against the year that country hosted the summer olympics.

Historical data,from the last six Summer Olympiads demonstrates a clear Host Nation Bumpfor every country. As shown in the provided visualizations, countries consistently achieve their highest medal totals in the 21st century during the years they host the Games (highlighted in gold). This trend suggests that Team USA is positioned for a record-breaking performance in Los Angeles 2028.

# 1. Define a function to simulate 10,000 outcomes for a specific medal type
simulate_medals <- function(medal_name, baseline_df, bump_df, new_sport_df) {
  set.seed(123) # For reproducibility
  n_sims <- 10000
  
  # A. Sample Baseline (Normal distribution based on your CI)
  # I use the estimate as the mean and derive standard error from the CI width
  b_est <- baseline_df |> filter(medal_type == medal_name) |> pull(estimate)
  b_se  <- (baseline_df |> filter(medal_type == medal_name) |> pull(upper_ci) - b_est) / 1.96
  sim_baseline <- rnorm(n_sims, mean = b_est, sd = b_se)
  
  # B. Sample Host Bump (Normal distribution)
  h_est <- bump_df |> filter(medal_type == medal_name) |> pull(estimate)
  h_se  <- (bump_df |> filter(medal_type == medal_name) |> pull(upper_ci) - h_est) / 1.96
  sim_bump <- rnorm(n_sims, mean = h_est, sd = h_se)
  
  # C. Sample New Sport Wins (Binomial distribution)
  # 22 is the number of new events for LA 2028
  p_new <- new_sport_df |> filter(medal_type == medal_name) |> pull(estimate)
  sim_new_sports <- rbinom(n_sims, size = 22, prob = p_new)
  
  # Baseline * (1 + Bump) + New Sport Wins
  return(sim_baseline * (1 + sim_bump) + sim_new_sports)
}

simulate_medals_safe <- function(medal_name, baseline_df, bump_df, new_sport_df) {
  set.seed(123)
  n_sims <- 10000
  
  # Helper to get values safely
  get_val <- function(df, m_name, col) {
    val <- df |> filter(medal_type == m_name) |> pull(!!sym(col))
    if(length(val) == 0) return(0) else return(val[1])
  }

  b_est <- get_val(baseline_df, medal_name, "estimate")
  b_se  <- (get_val(baseline_df, medal_name, "upper_ci") - b_est) / 1.96
  sim_baseline <- rnorm(n_sims, mean = b_est, sd = max(0.1, b_se))
  
  h_est <- get_val(bump_df, medal_name, "estimate")
  h_se  <- (get_val(bump_df, medal_name, "upper_ci") - h_est) / 1.96
  sim_bump <- rnorm(n_sims, mean = h_est, sd = max(0.1, h_se))
  
  p_new <- get_val(new_sport_df, medal_name, "estimate")
  sim_new_sports <- rbinom(n_sims, size = 22, prob = max(0, p_new))
  
  return(sim_baseline * (1 + sim_bump) + sim_new_sports)
}

#Hypotheses Testing

Null hypothesis H0: There is no difference in the average number of medals won between Host years and Non-Host years

Code 
# 1. (Hypothesis Testing) Calculate the observed difference in means
observed_diff <- statistical_data |>
  specify(total_medals ~ is_host) |>
  calculate(stat = "diff in means", order = c("Host", "Non-Host"))

# 2. Generate the null distribution (shuffling the labels 1000 times)
null_distribution <- statistical_data |>
  specify(total_medals ~ is_host) |>
  hypothesize(null = "independence") |>
  generate(reps = 1000, type = "permute") |>
  calculate(stat = "diff in means", order = c("Host", "Non-Host"))

# 3. Visualize the results
visualize(null_distribution) +
  shade_p_value(obs_stat = observed_diff, direction = "greater")
